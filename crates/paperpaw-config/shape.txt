# The shape of this crate's types and the values of its constants, as
# tools/interface.py writes them: fields, variants and trait items in
# the source's order, private fields and the private types they reach
# too, every feature on (ADR 0026).
enum paperpaw_config::ErrorKind { InvalidLine, InvalidSection, InvalidKey, UnterminatedString, InvalidEscape }
struct paperpaw_config::Entry<'a> { section: &'a str, key: &'a str, value: paperpaw_config::Value<'a>, line: u32 }
struct paperpaw_config::ParseError { line: u32, kind: paperpaw_config::ErrorKind }
struct paperpaw_config::Parser<'a> { lines: core::str::iter::Lines<'a>, line: u32, section: &'a str }
struct paperpaw_config::Value<'a> { raw: &'a str, quoted: bool }
